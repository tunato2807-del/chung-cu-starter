# Tài khoản căn hộ và bàn giao cho chủ hộ trong VS Code

Cập nhật sau commit `85578b8` (đã có điện bậc thang, chọn ngân hàng và QR thanh toán). Không cần tải ZIP: tạo/thay đủ **13 file** dưới đây. Giữ file banks.js và CSS của bản giao diện thanh toán trước. Không có dependency mới.

## Cách hoạt động

1. Tạo căn hộ sẽ tạo cùng lúc một tài khoản CU_DAN riêng. Username cố định theo ID căn, ví dụ **CH000001**, không dùng số điện thoại giả và không thay đổi khi sửa số căn. Mật khẩu ngẫu nhiên hiện một lần cho BQL để lưu/bàn giao; có thể đăng nhập ngay trước khi gắn hồ sơ.
2. Khi thêm cư dân hoặc ghi nhận nhập cư/chuyển căn với quan hệ **Chủ hộ**, hệ thống tự gắn tài khoản của căn vào hồ sơ người phụ trách thanh toán. Không cần tự tạo ở Nhân sự. Liên kết này riêng với tài khoản cá nhân tùy chọn; tài khoản cá nhân và số điện thoại hiện có không bị ghi đè.
3. Nếu căn đã có người phụ trách khác, cần chọn **Bàn giao tài khoản nếu căn đang có người phụ trách**. Không tự thay người đang phụ trách khi chưa chọn. Chỉ có một người phụ trách tài khoản căn ở một thời điểm, kể cả khi có nhiều hồ sơ đồng chủ hộ; hồ sơ/vai trò cư trú người cũ vẫn được giữ.
4. Bàn giao hoặc cấp lại mật khẩu sẽ tạo mật khẩu mới, ngắt mọi phiên cũ và lưu BQL/ngày thực hiện trong lịch sử. Người phụ trách chuyển đi thì tài khoản căn cũ bị tạm khóa. Khi chuyển làm chủ hộ căn mới, nhận tài khoản riêng của căn mới; không mang tài khoản căn cũ đi.
5. Tài khoản căn chỉ xem hóa đơn đã gửi/đã thu đúng căn, kể cả kỳ trước ngày người phụ trách mới nhập cư để đối soát sổ thu. Hóa đơn hiển thị **Người phụ trách thanh toán hiện tại**. Không tự đổi snapshot tài chính hoặc nghĩa vụ pháp lý của hóa đơn cũ; BQL vẫn xác nhận đã thu sau đối soát ngân hàng. Tài khoản cá nhân giữ quyền theo đợt cư trú như trước.

## Căn đã có dữ liệu trước bản cập nhật

Migration 007 tạo bổ sung một tài khoản khóa cho mỗi căn hiện có, không tạo lại cư dân và không đổi tài khoản cá nhân/đơn giá/hóa đơn. Nếu căn có đúng một chủ hộ đang ở thì tự gắn người đó làm phụ trách; nếu không có hoặc có nhiều chủ hộ, để BQL chọn. Các tài khoản này chưa có mật khẩu dùng được.

Sau cập nhật, BQL vào **Căn hộ → Xem → Tài khoản thanh toán căn hộ** để **Cấp lại mật khẩu và ngắt phiên cũ**; nếu cần thì chọn chủ hộ đang ở để **Bàn giao tài khoản**. Tài khoản căn được quản lý tại đây, không dùng nút khóa/mở tài khoản cá nhân ở Nhân sự để kích hoạt lại mật khẩu cũ.

Các mật khẩu mới chỉ trong phản hồi ghi của BQL và React state, không lưu localStorage/sessionStorage, không xem lại bằng API GET. Nếu đóng trang/lỡ ẩn thông tin, BQL cấp lại mật khẩu. Sau khi nhận, chủ hộ nên đổi mật khẩu ở **Tài khoản**.

```js
import bcrypt from "bcryptjs";
import { randomBytes } from "node:crypto";
import { pool, rows, transaction } from "../../lib/db.js";
import { fail, integer } from "../../lib/http.js";
import { lockApartment } from "../billing/billing.guards.js";

const columns =
  "u.ma_nguoi_dung,u.ten_dang_nhap,u.ma_can_ho_thanh_toan,u.ma_cu_dan_phu_trach,u.dang_hoat_dong,u.phien_ban_ban_giao,c.ho_va_ten AS nguoi_phu_trach,c.so_dien_thoai AS dien_thoai_phu_trach";
export async function accountForApartment(id, db = pool) {
  const [account] = await rows(
    db,
    `SELECT ${columns} FROM nguoi_dung u LEFT JOIN cu_dan c ON c.ma_cu_dan=u.ma_cu_dan_phu_trach WHERE u.ma_can_ho_thanh_toan=?`,
    [id],
  );
  return account || null;
}
async function audit(db, apartmentId, old, next, actor, event) {
  await rows(
    db,
    "INSERT INTO lich_su_ban_giao_can_ho (ma_can_ho,ma_cu_dan_cu,ma_cu_dan_moi,ma_nguoi_thuc_hien,su_kien) VALUES (?,?,?,?,?)",
    [apartmentId, old, next, actor.ma_nguoi_dung, event],
  );
}
async function rotate(db, account, apartment, owner, actor, event) {
  const secret = randomBytes(18).toString("base64url");
  const changingOwner = owner !== account.ma_cu_dan_phu_trach;
  await rows(
    db,
    `UPDATE nguoi_dung SET mat_khau_ma_hoa=?,dang_hoat_dong=1,${changingOwner ? "ma_cu_dan_phu_trach=?," : ""}ho_va_ten=?,phien_ban_token=phien_ban_token+1,phien_ban_ban_giao=phien_ban_ban_giao+1,so_lan_dang_nhap_sai=0,khoa_den_luc=NULL WHERE ma_nguoi_dung=?`,
    [
      await bcrypt.hash(secret, 12),
      ...(changingOwner ? [owner] : []),
      `Tài khoản căn ${apartment.so_can_ho}`,
      account.ma_nguoi_dung,
    ],
  );
  await audit(
    db,
    apartment.ma_can_ho,
    account.ma_cu_dan_phu_trach,
    owner,
    actor,
    event,
  );
  return {
    ...(await accountForApartment(apartment.ma_can_ho, db)),
    mat_khau_tam_thoi: secret,
  };
}
// Caller holds the apartment lock; generated credentials are returned once only.
export async function ensureApartmentAccount(db, apartment, actor) {
  const [existing] = await rows(
    db,
    "SELECT * FROM nguoi_dung WHERE ma_can_ho_thanh_toan=? FOR UPDATE",
    [apartment.ma_can_ho],
  );
  if (existing) return { account: existing, credentials: null };
  const username = `CH${String(apartment.ma_can_ho).padStart(6, "0")}`;
  const secret = randomBytes(18).toString("base64url");
  const r = await rows(
    db,
    "INSERT INTO nguoi_dung (ho_va_ten,ten_dang_nhap,so_dien_thoai,vai_tro,mat_khau_ma_hoa,ma_can_ho_thanh_toan) VALUES (?,?,NULL,'CU_DAN',?,?)",
    [
      `Tài khoản căn ${apartment.so_can_ho}`,
      username,
      await bcrypt.hash(secret, 12),
      apartment.ma_can_ho,
    ],
  );
  await audit(db, apartment.ma_can_ho, null, null, actor, "TAO_TAI_KHOAN");
  const [account] = await rows(
    db,
    "SELECT * FROM nguoi_dung WHERE ma_nguoi_dung=?",
    [r.insertId],
  );
  return {
    account,
    credentials: {
      ...(await accountForApartment(apartment.ma_can_ho, db)),
      mat_khau_tam_thoi: secret,
    },
  };
}
export async function assignHouseholder(
  db,
  apartment,
  residentId,
  actor,
  confirmed = false,
) {
  const { account } = await ensureApartmentAccount(db, apartment, actor);
  if (account.ma_cu_dan_phu_trach === residentId) return null;
  if (account.ma_cu_dan_phu_trach !== null && confirmed !== true)
    fail(
      409,
      "HANDOVER_REQUIRED",
      "Căn đã có người phụ trách thanh toán. Chọn xác nhận bàn giao tài khoản cho chủ hộ mới.",
    );
  return rotate(db, account, apartment, residentId, actor, "BAN_GIAO_CHU_HO");
}
export async function releaseHouseholder(db, apartmentId, residentId, actor) {
  const [account] = await rows(
    db,
    "SELECT * FROM nguoi_dung WHERE ma_can_ho_thanh_toan=? FOR UPDATE",
    [apartmentId],
  );
  if (!account || account.ma_cu_dan_phu_trach !== residentId) return;
  await rows(
    db,
    "UPDATE nguoi_dung SET ma_cu_dan_phu_trach=NULL,dang_hoat_dong=0,phien_ban_token=phien_ban_token+1,phien_ban_ban_giao=phien_ban_ban_giao+1 WHERE ma_nguoi_dung=?",
    [account.ma_nguoi_dung],
  );
  await audit(db, apartmentId, residentId, null, actor, "NGUNG_PHU_TRACH");
}
export async function manageApartmentAccount(id, body, actor) {
  return transaction(async (db) => {
    // Match residency's resident -> apartment -> account lock order. Only the
    // new owner needs a resident FK lock; audit IDs are immutable snapshots.
    const residentId =
      body.thao_tac === "ban_giao"
        ? integer(body.ma_cu_dan, "ma_cu_dan")
        : null;
    if (residentId !== null)
      await rows(
        db,
        "SELECT ma_cu_dan FROM cu_dan WHERE ma_cu_dan=? FOR UPDATE",
        [residentId],
      );
    const apartment = await lockApartment(db, id);
    const { account } = await ensureApartmentAccount(db, apartment, actor);
    if (
      integer(body.phien_ban_ban_giao, "phien_ban_ban_giao") !==
      account.phien_ban_ban_giao
    )
      fail(
        409,
        "STALE_VERSION",
        "Thông tin tài khoản đã thay đổi. Hãy tải lại.",
      );
    if (body.thao_tac === "cap_lai_mat_khau")
      return rotate(
        db,
        account,
        apartment,
        account.ma_cu_dan_phu_trach,
        actor,
        "CAP_LAI_MAT_KHAU",
      );
    if (body.thao_tac !== "ban_giao")
      fail(422, "INVALID_ACTION", "Thao tác tài khoản không hợp lệ.");
    // All stay changes lock this apartment before mutation. Read after that lock,
    // without taking resident locks in the reverse order of the residency flow.
    const [owner] = await rows(
      db,
      "SELECT ma_cu_dan FROM lich_su_cu_tru WHERE ma_cu_dan=? AND ma_can_ho=? AND quan_he='CHU_HO' AND den_ngay IS NULL",
      [residentId, id],
    );
    if (!owner)
      fail(
        422,
        "NOT_HOUSEHOLDER",
        "Chỉ bàn giao cho cư dân đang là chủ hộ của căn này.",
      );
    if (account.ma_cu_dan_phu_trach === residentId)
      fail(
        409,
        "ALREADY_RESPONSIBLE",
        "Cư dân này đã phụ trách. Dùng Cấp lại mật khẩu nếu cần.",
      );
    return assignHouseholder(
      db,
      apartment,
      residentId,
      actor,
      body.xac_nhan_ban_giao,
    );
  });
}
```

## 5. `backend/src/modules/auth/auth.service.js`

Thay toàn bộ nội dung file.

```js
import bcrypt from "bcryptjs";
import { SignJWT, jwtVerify } from "jose";
import { transaction, rows } from "../../lib/db.js";
import { fail, phone, text, choice } from "../../lib/http.js";
import { findUser, publicColumns } from "./auth.repository.js";
const secret = () => {
  if (!process.env.JWT_SECRET || process.env.JWT_SECRET.length < 32)
    throw new Error(
      "JWT_SECRET must be at least 32 characters. Run npm run init:local.",
    );
  return new TextEncoder().encode(process.env.JWT_SECRET);
};
const dummyHash = bcrypt.hashSync("dummy-password-never-used", 12);
export function password(value, field = "mat_khau") {
  // bcrypt only processes 72 bytes. Reject rather than silently truncate.
  if (
    typeof value !== "string" ||
    value.length < 10 ||
    Buffer.byteLength(value) > 72
  )
    fail(422, "VALIDATION", "Mật khẩu cần từ 10 ký tự và không quá 72 byte.", {
      [field]: "Cần từ 10 ký tự, tối đa 72 byte.",
    });
  return value;
}
export async function login(body) {
  const identifier = text(
    body.ten_dang_nhap ?? body.so_dien_thoai,
    "ten_dang_nhap",
    1,
    30,
  ).trim();
  const apartmentLogin = /^CH\d{6,10}$/i.test(identifier);
  const number = apartmentLogin ? identifier.toUpperCase() : phone(identifier);
  if (
    typeof body.mat_khau !== "string" ||
    Buffer.byteLength(body.mat_khau) > 72
  )
    fail(
      401,
      "LOGIN_FAILED",
      "Thông tin đăng nhập không đúng hoặc tài khoản tạm khóa.",
    );
  const user = await transaction(async (db) => {
    const [u] = await rows(
      db,
      `SELECT * FROM nguoi_dung WHERE ${apartmentLogin ? "ten_dang_nhap" : "so_dien_thoai"}=? FOR UPDATE`,
      [number],
    );
    const valid = await bcrypt.compare(
      body.mat_khau,
      u?.mat_khau_ma_hoa || dummyHash,
    );
    if (
      !u ||
      !u.dang_hoat_dong ||
      (u.khoa_den_luc && new Date(u.khoa_den_luc + "Z").getTime() > Date.now())
    )
      return null;
    if (!valid) {
      await rows(
        db,
        "UPDATE nguoi_dung SET so_lan_dang_nhap_sai=so_lan_dang_nhap_sai+1, khoa_den_luc=IF(so_lan_dang_nhap_sai>=5,DATE_ADD(UTC_TIMESTAMP(),INTERVAL 15 MINUTE),NULL) WHERE ma_nguoi_dung=?",
        [u.ma_nguoi_dung],
      );
      return null;
    }
    await rows(
      db,
      "UPDATE nguoi_dung SET so_lan_dang_nhap_sai=0,khoa_den_luc=NULL WHERE ma_nguoi_dung=?",
      [u.ma_nguoi_dung],
    );
    return findUser(u.ma_nguoi_dung, db);
  });
  if (!user)
    fail(
      401,
      "LOGIN_FAILED",
      "Thông tin đăng nhập không đúng hoặc tài khoản tạm khóa.",
    );
  const token = await new SignJWT({ version: user.phien_ban_token })
    .setProtectedHeader({ alg: "HS256" })
    .setSubject(String(user.ma_nguoi_dung))
    .setIssuer("chung-cu")
    .setAudience("chung-cu-web")
    .setIssuedAt()
    .setExpirationTime("8h")
    .sign(secret());
  return { user, token };
}
export async function authenticate(token) {
  try {
    const { payload } = await jwtVerify(token, secret(), {
      algorithms: ["HS256"],
      issuer: "chung-cu",
      audience: "chung-cu-web",
    });
    if (!/^\d+$/.test(payload.sub)) return null;
    const u = await findUser(Number(payload.sub));
    return u?.dang_hoat_dong && u.phien_ban_token === payload.version
      ? u
      : null;
  } catch (error) {
    if (error.code?.startsWith("ERR_J")) return null;
    throw error;
  }
}
export async function changePassword(user, body) {
  const fresh = password(body.mat_khau_moi, "mat_khau_moi");
  if (
    typeof body.mat_khau_cu !== "string" ||
    Buffer.byteLength(body.mat_khau_cu) > 72
  )
    fail(422, "PASSWORD", "Mật khẩu hiện tại không đúng.");
  await transaction(async (db) => {
    const [u] = await rows(
      db,
      "SELECT * FROM nguoi_dung WHERE ma_nguoi_dung=? FOR UPDATE",
      [user.ma_nguoi_dung],
    );
    if (!(await bcrypt.compare(body.mat_khau_cu, u.mat_khau_ma_hoa)))
      fail(422, "PASSWORD", "Mật khẩu hiện tại không đúng.");
    await rows(
      db,
      "UPDATE nguoi_dung SET mat_khau_ma_hoa=?,phien_ban_token=phien_ban_token+1 WHERE ma_nguoi_dung=?",
      [await bcrypt.hash(fresh, 12), user.ma_nguoi_dung],
    );
  });
}
export async function createUser(body) {
  const name = text(body.ho_va_ten, "ho_va_ten", 2, 100),
    number = phone(body.so_dien_thoai);
  const role = choice(body.vai_tro, "vai_tro", ["KY_THUAT_VIEN", "CU_DAN"]);
  const hash = await bcrypt.hash(password(body.mat_khau), 12);
  return transaction(async (db) => {
    const r = await rows(
      db,
      "INSERT INTO nguoi_dung (ho_va_ten,so_dien_thoai,vai_tro,mat_khau_ma_hoa,chuyen_mon) VALUES (?,?,?,?,?)",
      [
        name,
        number,
        role,
        hash,
        text(body.chuyen_mon || "", "chuyen_mon", 0, 100),
      ],
    );
    return findUser(r.insertId, db);
  });
}
export async function setUserActive(actor, id, active) {
  if (id === actor.ma_nguoi_dung)
    fail(422, "SELF_LOCK", "Không thể tự khóa tài khoản.");
  if (typeof active !== "boolean")
    fail(422, "VALIDATION", "Trạng thái tài khoản không hợp lệ.");
  return transaction(async (db) => {
    const [user] = await rows(
      db,
      `SELECT ${publicColumns} FROM nguoi_dung WHERE ma_nguoi_dung=? FOR UPDATE`,

    );
```

```js
          409,
          "ALREADY_RESIDENT",
          "Cư dân đang ở một căn hộ. Hãy dùng Chuyển căn.",
        );

        fail(

          "OVERLAPPING_STAY",
          "Ngày nhập cư phải từ ngày kết thúc đợt cư trú gần nhất.",

          "Ngày chuyển không được trước ngày bắt đầu đợt cư trú hiện tại.",
        );
      if (next && next[0] === current.ma_can_ho)
        fail(422, "SAME_APARTMENT", "Cần chọn căn hộ khác để chuyển căn.");
    }
    if (next) await one(db, "can_ho", "ma_can_ho", next[0]);
    for (const apartmentId of [
      ...new Set([current?.ma_can_ho, next?.[0]].filter(Boolean)),
    ].sort((a, b) => a - b))
      await lockApartment(db, apartmentId);
    const vehicles = current
      ? await finishStayVehicles(
          db,
          current.ma_lich_su_cu_tru,
          day,
          user.ma_nguoi_dung,

        )
      : [];
    if (current) await releaseHouseholder(db, current.ma_can_ho, id, user);
    if (current)
      await rows(
        db,
        "UPDATE lich_su_cu_tru SET den_ngay=? WHERE ma_lich_su_cu_tru=?",
        [day, current.ma_lich_su_cu_tru],

          user,
```

```js
    `SELECT m.*,a.so_can_ho FROM chi_so_dich_vu m JOIN can_ho a ON a.ma_can_ho=m.ma_can_ho ${condition} ORDER BY m.ky_thanh_toan DESC,m.ma_chi_so_dich_vu DESC LIMIT 20 OFFSET ${offset}`,
    values,
  );
  return { data, total: count.total, page };
}
export async function saveMeter(body, user) {
  const apartmentId = integer(body.ma_can_ho, "ma_can_ho"),
    type = choice(body.loai_dich_vu, "loai_dich_vu", ["DIEN", "NUOC"]),
    month = v.period(body.ky_thanh_toan),
    day = v.monthDay(body.ngay_chot, month);
  const old = v.scaled(body.chi_so_cu, 3, 9),
    next = v.scaled(body.chi_so_moi, 3, 9),
    note = text(body.ghi_chu ?? "", "ghi_chu", 0, 255);
  if (next < old)
    fail(422, "METER_DECREASE", "Chỉ số mới không được nhỏ hơn chỉ số cũ.");
  return transaction(async (db) => {
    await lockApartment(db, apartmentId);
    const billed = await rows(
      db,
      "SELECT ma_hoa_don FROM hoa_don WHERE ma_can_ho=? AND ky_thanh_toan=? AND trang_thai IN ('DA_DUYET','DA_GUI','DA_THANH_TOAN') LIMIT 1 FOR UPDATE",
      [apartmentId, month],
    );
    if (billed.length)
      fail(
        409,
        "METER_LOCKED",
        "Kỳ này đã có hóa đơn được duyệt. Hủy hóa đơn chưa thanh toán trước khi điều chỉnh chỉ số.",
      );
    const records = await rows(
      db,
      "SELECT * FROM chi_so_dich_vu WHERE ma_can_ho=? AND loai_dich_vu=? ORDER BY ky_thanh_toan DESC FOR UPDATE",
      [apartmentId, type],
    );
    if (records[0]?.ky_thanh_toan > month)
      fail(
        409,
        "METER_LATER_PERIOD",
        "Chỉ ghi hoặc sửa kỳ mới nhất; đã có chỉ số kỳ sau.",
      );
    const current = records.find((r) => r.ky_thanh_toan === month),
      previous = records.find((r) => r.ky_thanh_toan < month);
    if (previous) {
      const prior = new Date(`${month}-01T00:00:00Z`);
      prior.setUTCMonth(prior.getUTCMonth() - 1);
      if (previous.ky_thanh_toan !== prior.toISOString().slice(0, 7))
        fail(
          422,
          "MISSING_PREVIOUS_METER",
          "Thiếu chỉ số kỳ liền trước. Hãy bổ sung trước khi ghi kỳ này.",
        );
      if (old !== v.scaled(previous.chi_so_moi, 3, 9))
        fail(
          422,
          "METER_CHAIN",
          "Chỉ số cũ phải bằng chỉ số mới của kỳ liền trước.",
        );
    } else if (!current && body.ky_dau !== true)
      fail(
        422,
        "FIRST_METER",
        "Kỳ đầu cần xác nhận nhập cả chỉ số đầu và cuối.",
      );
    if (current) {
      v.version(current, body.phien_ban);
      await rows(
        db,
        "UPDATE chi_so_dich_vu SET chi_so_cu=?,chi_so_moi=?,ngay_chot=?,ghi_chu=?,ma_nguoi_ghi=?,ngay_ghi=CURRENT_TIMESTAMP,phien_ban=phien_ban+1 WHERE ma_chi_so_dich_vu=?",
        [
          v.decimal(old, 3),
          v.decimal(next, 3),
          day,
          note,
          user.ma_nguoi_dung,
          current.ma_chi_so_dich_vu,
        ],
      );
      return { ma_chi_so_dich_vu: current.ma_chi_so_dich_vu };
    }
    const result = await rows(
      db,
      "INSERT INTO chi_so_dich_vu (ma_can_ho,loai_dich_vu,ky_thanh_toan,chi_so_cu,chi_so_moi,ngay_chot,ghi_chu,ma_nguoi_ghi) VALUES (?,?,?,?,?,?,?,?)",
      [
        apartmentId,
        type,
        month,
        v.decimal(old, 3),
        v.decimal(next, 3),
        day,
        note,
        user.ma_nguoi_dung,
      ],
    );
    return { ma_chi_so_dich_vu: result.insertId };
  });
}

async function calculate(db, apartment, month, day, options) {
  await rows(db, "SELECT loai_phi FROM loai_phi ORDER BY loai_phi FOR UPDATE");
  const rates = await rows(
    db,
    "SELECT c.*,l.ten_phi,l.don_vi_tinh FROM cau_hinh_phi c JOIN loai_phi l ON l.loai_phi=c.loai_phi WHERE c.tu_ngay<=? AND (c.den_ngay IS NULL OR c.den_ngay>?) ORDER BY c.loai_phi FOR UPDATE",
    [day, day],
  );
  const meters = await rows(
    db,
    "SELECT * FROM chi_so_dich_vu WHERE ma_can_ho=? AND ky_thanh_toan=? ORDER BY loai_dich_vu FOR UPDATE",
    [apartment.ma_can_ho, month],
  );
  const cars = options.tinh_xe
    ? await rows(
        db,
        "SELECT ma_phuong_tien,loai_xe FROM phuong_tien WHERE ma_can_ho=? AND ngay_dang_ky<=? AND (ngay_ket_thuc IS NULL OR ngay_ket_thuc>?) ORDER BY ma_phuong_tien FOR UPDATE",
        [apartment.ma_can_ho, day, day],
      )
    : [];
  const lines = [],
    errors = [];
  function add(type, quantity, source = {}, meterId = null) {
    const rate = rates.find((r) => r.loai_phi === type);
    if (!rate) {
      errors.push(`Thiếu đơn giá ${type} tại ngày chốt.`);
      return;
    }
    const qty = v.decimal(v.scaled(quantity, 3, 9), 3),
      price = v.decimal(v.scaled(rate.don_gia), 2);
    lines.push({
      loai_phi: type,
      dien_giai: rate.ten_phi,
      don_vi_tinh: rate.don_vi_tinh,
      so_luong: qty,
      don_gia: price,
      thanh_tien: v.lineTotal(qty, price),
      ma_cau_hinh_phi: rate.ma_cau_hinh_phi,
      ma_chi_so_dich_vu: meterId,
      du_lieu_nguon: source,
    });
  }
  if (apartment.tinh_phi_dich_vu) add("PHI_QUAN_LY", apartment.dien_tich_m2);
  if (options.tinh_xe)
    for (const [type, fee] of [
      ["O_TO", "GUI_O_TO"],
      ["XE_MAY", "GUI_XE_MAY"],
      ["XE_DAP", "GUI_XE_DAP"],
    ]) {
      const selected = cars.filter((r) => r.loai_xe === type);
      if (selected.length)
        add(fee, String(selected.length), {
          ma_dang_ky: selected.map((r) => r.ma_phuong_tien),
        });
    }
  for (const [type, enabled] of [
    ["NUOC", options.tinh_nuoc],
    ["DIEN", options.tinh_dien],
  ])
    if (enabled) {
      const meter = meters.find((r) => r.loai_dich_vu === type);
      if (!meter || meter.ngay_chot !== day) {
        errors.push(`Thiếu chỉ số ${type} đúng ngày chốt ${day}.`);
        continue;
      }
      const used =
        v.scaled(meter.chi_so_moi, 3, 9) - v.scaled(meter.chi_so_cu, 3, 9);
      if (type === "DIEN") {
        const rate = rates.find((r) => r.loai_phi === "TIEN_DIEN");
        if (!rate) errors.push("Thiếu đơn giá TIEN_DIEN tại ngày chốt.");
        else {
          // Preserve legacy flat/no-VAT snapshots byte-for-byte for existing drafts.
          if (rate.cach_tinh === "BAC_THANG" || v.scaled(rate.thue_gtgt) > 0n) {
            lines.push(
              ...electricityLines(
                rate,
                used,
                {
                  chi_so_cu: meter.chi_so_cu,
                  chi_so_moi: meter.chi_so_moi,
                  phien_ban: meter.phien_ban,
                },
                meter.ma_chi_so_dich_vu,
              ),
            );
            continue;
          }
        }
        if (!rate) continue;
      }
      add(
        `TIEN_${type}`,
        v.decimal(used, 3),
        {
          chi_so_cu: meter.chi_so_cu,
          chi_so_moi: meter.chi_so_moi,
          phien_ban: meter.phien_ban,
        },
        meter.ma_chi_so_dich_vu,
      );
    }
  if (!lines.length) errors.push("Chưa có khoản phí hợp lệ được áp dụng.");
  const total = lines.reduce((n, l) => n + BigInt(l.thanh_tien), 0n);
  if (total > 99999999999999n)
    fail(422, "AMOUNT_TOO_LARGE", "Tổng hóa đơn vượt giới hạn.");
  return {
    lines,
    errors,
    total: total.toString(),
    hash: digest({ lines, errors }),
  };
}
async function storeSnapshot(db, id, snapshot) {
  await rows(db, "DELETE FROM chi_tiet_hoa_don WHERE ma_hoa_don=?", [id]);
  for (const l of snapshot.lines)
    await rows(
      db,
      "INSERT INTO chi_tiet_hoa_don (ma_hoa_don,ma_cau_hinh_phi,ma_chi_so_dich_vu,loai_phi,dien_giai,don_vi_tinh,so_luong,don_gia,thanh_tien,du_lieu_nguon) VALUES (?,?,?,?,?,?,?,?,?,?)",
      [
        id,
        l.ma_cau_hinh_phi,
        l.ma_chi_so_dich_vu,
        l.loai_phi,
        l.dien_giai,
        l.don_vi_tinh,
        l.so_luong,
        l.don_gia,
        l.thanh_tien,
        JSON.stringify(l.du_lieu_nguon),
      ],
    );
  await rows(
    db,
    "UPDATE hoa_don SET tong_tien=?,loi_du_lieu=?,dau_van_tay=? WHERE ma_hoa_don=?",
    [snapshot.total, JSON.stringify(snapshot.errors), snapshot.hash, id],
  );
}
async function makeDraft(db, apartment, body, user, previous = null) {
  const month = v.period(body.ky_thanh_toan),
    day = v.monthDay(body.ngay_chot_phi, month),
    options = v.options(body);
  const existing = await rows(
    db,
    "SELECT * FROM hoa_don WHERE ma_can_ho=? AND ky_thanh_toan=? ORDER BY lan_lap DESC FOR UPDATE",
    [apartment.ma_can_ho, month],
  );
  if (existing.some((h) => h.trang_thai !== "HUY"))
    fail(
      409,
      "BILL_EXISTS",
      "Căn đã có hóa đơn còn hiệu lực trong kỳ. Mở hóa đơn nháp để tính lại.",
    );
  if (
    existing.length &&
    (!previous || existing[0].ma_hoa_don !== previous.ma_hoa_don)
  )
    fail(
      409,
      "REPLACEMENT_REQUIRED",
      "Hãy lập thay thế từ hóa đơn hủy gần nhất.",
    );
  const snapshot = await calculate(db, apartment, month, day, options);
  const result = await rows(
    db,
    "INSERT INTO hoa_don (so_hoa_don,ma_can_ho,ky_thanh_toan,ngay_chot_phi,lan_lap,tinh_nuoc,tinh_dien,tinh_xe,loi_du_lieu,dau_van_tay,ma_nguoi_lap,ma_hoa_don_thay_the_cho) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
    [
      `HD${month.replace("-", "")}-${randomBytes(6).toString("hex").toUpperCase()}`,
      apartment.ma_can_ho,
      month,
      day,
      (existing[0]?.lan_lap || 0) + 1,
      options.tinh_nuoc,
      options.tinh_dien,
      options.tinh_xe,
      "[]",
      snapshot.hash,
      user.ma_nguoi_dung,
      previous?.ma_hoa_don || null,
    ],
  );
  await storeSnapshot(db, result.insertId, snapshot);
  await log(db, result.insertId, user, previous ? "LAP_THAY_THE" : "LAP_NHAP");
  return { ma_hoa_don: result.insertId };
}
export async function generate(body, user) {
  const id = integer(body.ma_can_ho, "ma_can_ho");
  return transaction(async (db) =>
    makeDraft(db, await lockApartment(db, id), body, user),
  );
}
async function lockedBill(db, id) {
  const [row] = await rows(
    db,
    "SELECT ma_can_ho FROM hoa_don WHERE ma_hoa_don=?",
    [id],
  );
  if (!row) fail(404, "NOT_FOUND", "Không tìm thấy hóa đơn.");
  const apartment = await lockApartment(db, row.ma_can_ho);
  const [bill] = await rows(
    db,
    "SELECT * FROM hoa_don WHERE ma_hoa_don=? FOR UPDATE",
    [id],
  );
  return { bill, apartment };
}
export async function action(id, action, body, user) {
  return transaction(async (db) => {
    const { bill, apartment } = await lockedBill(db, id);
    if (
      action === "xac_nhan_thanh_toan" &&
      bill.trang_thai === "DA_THANH_TOAN"
    ) {
      if (
        v.amount(body.so_tien).toString() === bill.so_tien_da_thu &&
        text(body.ma_tham_chieu, "ma_tham_chieu", 3, 100) === bill.ma_tham_chieu
      )
        return { ma_hoa_don: id };
      fail(409, "ALREADY_PAID", "Hóa đơn đã được xác nhận thu.");
    }
    v.version(bill, body.phien_ban);
    if (action === "lap_thay_the") {
      state(bill, ["HUY"]);
      return makeDraft(
        db,
        apartment,
        {
          ky_thanh_toan: bill.ky_thanh_toan,
          ngay_chot_phi: body.ngay_chot_phi ?? bill.ngay_chot_phi,
          ...opts(bill, body),
        },
        user,
        bill,
      );
    }
    if (action === "tinh_lai") {
      state(bill, ["NHAP"]);
      const day = v.monthDay(
          body.ngay_chot_phi ?? bill.ngay_chot_phi,
          bill.ky_thanh_toan,
        ),
        options = opts(bill, body);
      await storeSnapshot(
        db,
        id,
        await calculate(db, apartment, bill.ky_thanh_toan, day, options),
      );
      await rows(
        db,
        "UPDATE hoa_don SET ngay_chot_phi=?,tinh_nuoc=?,tinh_dien=?,tinh_xe=? WHERE ma_hoa_don=?",
        [day, options.tinh_nuoc, options.tinh_dien, options.tinh_xe, id],
      );
    } else if (action === "duyet") {
      state(bill, ["NHAP"]);
      const current = await calculate(
        db,
        apartment,
        bill.ky_thanh_toan,
        bill.ngay_chot_phi,
        opts(bill),
      );
      if (current.errors.length || BigInt(current.total) <= 0n)
        fail(
          422,
          "INCOMPLETE_BILL",
          "Hóa đơn còn thiếu dữ liệu hoặc tổng bằng 0. Bổ sung và tính lại bản nháp.",
        );
      if (current.hash !== bill.dau_van_tay || current.total !== bill.tong_tien)
        fail(
          409,
          "STALE_BILL_SOURCE",
          "Dữ liệu nguồn đã thay đổi. Hãy tính lại bản nháp trước khi duyệt.",
        );
      const saved = await rows(
        db,
        "SELECT * FROM chi_tiet_hoa_don WHERE ma_hoa_don=? ORDER BY ma_chi_tiet_hoa_don",
        [id],
      );
      const lines = saved.map(({ ma_hoa_don, ma_chi_tiet_hoa_don, ...l }) => ({
        ...l,
        du_lieu_nguon: json(l.du_lieu_nguon),
      }));
      if (
        digest({ lines, errors: json(bill.loi_du_lieu) }) !== bill.dau_van_tay
      )
        fail(
          409,
          "BILL_TOTAL_MISMATCH",
          "Chi tiết hóa đơn không khớp dữ liệu đã lập. Hãy tính lại.",
        );
      await rows(
        db,
        "UPDATE hoa_don SET trang_thai='DA_DUYET',ma_nguoi_duyet=?,ngay_duyet=CURRENT_TIMESTAMP WHERE ma_hoa_don=?",
        [user.ma_nguoi_dung, id],
      );
    } else if (action === "gui") {
      state(bill, ["DA_DUYET"]);
      const due = v.day(body.han_thanh_toan, true),
        bin = text(body.ma_ngan_hang, "ma_ngan_hang", 6, 6),
        account = text(body.so_tai_khoan, "so_tai_khoan", 6, 30),
        name = text(body.ten_nguoi_nhan, "ten_nguoi_nhan", 2, 100);
      if (
        due < bill.ngay_chot_phi ||
        due < v.today() ||
        !/^\d{6}$/.test(bin) ||
        !/^\d{6,30}$/.test(account)
      )
        fail(
          422,
          "INVALID_PAYMENT_DETAILS",
          "Hạn trả phải từ hôm nay và không trước ngày chốt; BIN gồm 6 số, tài khoản gồm 6–30 số.",
        );
      await rows(
        db,
        "UPDATE hoa_don SET trang_thai='DA_GUI',han_thanh_toan=?,ma_ngan_hang=?,so_tai_khoan=?,ten_nguoi_nhan=?,ma_nguoi_gui=?,ngay_gui=CURRENT_TIMESTAMP WHERE ma_hoa_don=?",
        [due, bin, account, name, user.ma_nguoi_dung, id],
      );
    } else if (action === "huy") {
      state(bill, ["NHAP", "DA_DUYET", "DA_GUI"]);
      const reason = text(body.ly_do, "ly_do", 3, 255);
      await rows(
        db,
        "UPDATE hoa_don SET trang_thai='HUY',ly_do_huy=?,ma_nguoi_huy=?,ngay_huy=CURRENT_TIMESTAMP WHERE ma_hoa_don=?",
        [reason, user.ma_nguoi_dung, id],
      );
    } else if (action === "xac_nhan_thanh_toan") {
      state(bill, ["DA_GUI"]);
      const amount = v.amount(body.so_tien),
        reference = text(body.ma_tham_chieu, "ma_tham_chieu", 3, 100);
      if (amount !== BigInt(bill.tong_tien))
        fail(
          422,
          "WRONG_AMOUNT",
          "Chỉ xác nhận khi số tiền đã đối soát bằng đúng tổng hóa đơn.",
        );
      await rows(
        db,
        "UPDATE hoa_don SET trang_thai='DA_THANH_TOAN',so_tien_da_thu=?,ma_tham_chieu=?,ma_nguoi_xac_nhan=?,ngay_thanh_toan=CURRENT_TIMESTAMP WHERE ma_hoa_don=?",
        [amount.toString(), reference, user.ma_nguoi_dung, id],
      );
    } else fail(404, "NOT_FOUND", "Không có thao tác hóa đơn này.");
    await rows(
      db,
      "UPDATE hoa_don SET phien_ban=phien_ban+1 WHERE ma_hoa_don=?",
      [id],
    );
    await log(
      db,
      id,
      user,
      action.toUpperCase(),
      action === "huy" ? text(body.ly_do, "ly_do", 3, 255) : "",
    );
    return { ma_hoa_don: id };
  });
}

function access(user, where, values) {
  if (user.vai_tro === "QUAN_TRI_VIEN") return;
  if (user.vai_tro !== "CU_DAN")
    fail(403, "FORBIDDEN", "Bạn không có quyền xem hóa đơn.");
  if (user.ma_can_ho_thanh_toan !== null) {
    where.push(
      "h.trang_thai IN ('DA_GUI','DA_THANH_TOAN') AND EXISTS (SELECT 1 FROM nguoi_dung u WHERE u.ma_nguoi_dung=? AND u.ma_can_ho_thanh_toan=h.ma_can_ho AND u.dang_hoat_dong=1 AND u.phien_ban_token=?)",
    );
    values.push(user.ma_nguoi_dung, user.phien_ban_token);
  } else {
    where.push(
      "h.trang_thai IN ('DA_GUI','DA_THANH_TOAN') AND EXISTS (SELECT 1 FROM cu_dan c JOIN lich_su_cu_tru l ON l.ma_cu_dan=c.ma_cu_dan WHERE c.ma_nguoi_dung=? AND l.ma_can_ho=h.ma_can_ho AND l.den_ngay IS NULL AND l.tu_ngay<=h.ngay_chot_phi)",
    );
    values.push(user.ma_nguoi_dung);
  }
}
function summary(bill) {
  return {
    ...bill,
    qua_han: bill.trang_thai === "DA_GUI" && bill.han_thanh_toan < v.today(),
  };
}
export async function list(query, user) {
  const { page, offset } = v.pagination(query),
    where = [],
    values = [];
  access(user, where, values);
  if (query.ky_thanh_toan) {
    where.push("h.ky_thanh_toan=?");
    values.push(v.period(query.ky_thanh_toan));
  }
  if (query.trang_thai) {
    where.push("h.trang_thai=?");
    values.push(
      choice(query.trang_thai, "trang_thai", [
        "NHAP",
        "DA_DUYET",
        "DA_GUI",
        "DA_THANH_TOAN",
        "HUY",
      ]),
    );
  }
  if (query.ma_can_ho) {
    where.push("h.ma_can_ho=?");
    values.push(integer(query.ma_can_ho, "ma_can_ho"));
  }
  if (query.q) {
    const q = text(query.q, "q", 1, 100);
    where.push("(h.so_hoa_don LIKE ? OR a.so_can_ho LIKE ?)");
    values.push(`%${q}%`, `%${q}%`);
  }
  const condition = where.length ? `WHERE ${where.join(" AND ")}` : "";
  const [count] = await rows(
    pool,
    `SELECT COUNT(*) AS total FROM hoa_don h JOIN can_ho a ON a.ma_can_ho=h.ma_can_ho ${condition}`,
    values,
  );
  const data = await rows(
    pool,
    `SELECT h.ma_hoa_don,h.so_hoa_don,h.ky_thanh_toan,h.ngay_chot_phi,h.trang_thai,h.tong_tien,h.han_thanh_toan,a.so_can_ho FROM hoa_don h JOIN can_ho a ON a.ma_can_ho=h.ma_can_ho ${condition} ORDER BY h.ma_hoa_don DESC LIMIT 20 OFFSET ${offset}`,
    values,
  );
  return { data: data.map(summary), total: count.total, page };
}
export async function detail(id, user) {
  const where = ["h.ma_hoa_don=?"],
    values = [id];
  access(user, where, values);
  const [row] = await rows(
    pool,
    `SELECT h.*,a.so_can_ho,c.ho_va_ten AS nguoi_phu_trach_thanh_toan FROM hoa_don h JOIN can_ho a ON a.ma_can_ho=h.ma_can_ho LEFT JOIN nguoi_dung tk ON tk.ma_can_ho_thanh_toan=h.ma_can_ho LEFT JOIN cu_dan c ON c.ma_cu_dan=tk.ma_cu_dan_phu_trach WHERE ${where.join(" AND ")}`,
    values,
  );
  if (!row)
    fail(
      404,
      "NOT_FOUND",
```

```jsx
function BankSelect() {
  const [query, setQuery] = useState(""),
    [selected, setSelected] = useState("");
  const matches = banks.filter((bank) =>
    bankSearch(`${bank.shortName} ${bank.name} ${bank.bin}`).includes(
      bankSearch(query),
    ),
  );
  const current = banks.find((bank) => bank.bin === selected);
  const option = (bank) => (
    <option key={bank.bin} value={bank.bin}>
      {bank.shortName} — {bank.name}
    </option>
  );
  return (
    <div className="bank-picker">
      <Field label="Tìm ngân hàng">
        <input
          type="search"
          value={query}
          maxLength={100}
          placeholder="Nhập tên ngân hàng, ví dụ Vietcombank"
          onChange={(e) => setQuery(e.target.value)}
        />
      </Field>
      <Field label="Ngân hàng nhận tiền">
        <select
          name="ma_ngan_hang"
          required
          value={selected}
          onChange={(e) => setSelected(e.target.value)}
        >
          <option value="">Chọn ngân hàng</option>
          {current &&
            !matches.some((bank) => bank.bin === selected) &&
            option(current)}
          {matches.map(option)}
        </select>
      </Field>
      {!matches.length && (
        <p className="muted">
          Không tìm thấy ngân hàng. Hãy thử tên khác hoặc xóa từ khóa.
        </p>
      )}
      {current && (
        <p className="bank-selection">
          Đã chọn: <strong>{current.shortName}</strong>
        </p>
      )}
    </div>
  );
}
function PaymentQR({ bill }) {
  const [image, setImage] = useState(""),
    [error, setError] = useState(null);
  useEffect(() => {
    let active = true;
    setImage("");
    setError(null);
    if (bill.qr_payload)
      QRCode.toDataURL(bill.qr_payload, {
        width: 320,
        margin: 2,
        errorCorrectionLevel: "M",
      })
        .then((url) => {
          if (active) setImage(url);
        })
        .catch((e) => {
          if (active) setError(e);
        });
    return () => {
      active = false;
    };
  }, [bill.qr_payload]);
  if (!bill.qr_payload) return null;
  return (
    <section
      className="card payment-card"
      aria-label="Thông tin thanh toán hóa đơn"
    >
      <header className="payment-header">
        <span className="payment-status">Chờ thanh toán</span>
        <h2>Chuyển khoản bằng VietQR</h2>
        <p>
          Căn {bill.so_can_ho} · Kỳ {bill.ky_thanh_toan}
        </p>
      </header>
      <div className="payment-body">
        <p className="payment-amount-label">Số tiền cần thanh toán</p>
        <p className="payment-amount">{money(bill.tong_tien)}</p>
        <div className="payment-qr-frame">
          <span className="payment-qr-brand">VIETQR</span>
          {image ? (
            <img
              src={image}
              alt={`VietQR hóa đơn ${bill.so_hoa_don}`}
              width="320"
              height="320"
            />
          ) : (
            !error && <p role="status">Đang tạo mã QR…</p>
          )}
          <ErrorBox error={error} />
          <span className="payment-qr-caption">
            Quét mã bằng ứng dụng ngân hàng
          </span>
        </div>
        <dl className="payment-details">
          <div>
            <dt>Ngân hàng</dt>
            <dd>{bankName(bill.ma_ngan_hang)}</dd>
          </div>
          <div>
            <dt>Số tài khoản</dt>
            <dd className="payment-account">{bill.so_tai_khoan}</dd>
          </div>
          <div>
            <dt>Người nhận</dt>
            <dd>{bill.ten_nguoi_nhan}</dd>
          </div>
          <div className="payment-reference">
            <dt>Nội dung chuyển khoản</dt>
            <dd>{bill.so_hoa_don}</dd>
          </div>
          <div>
            <dt>Hạn thanh toán</dt>
            <dd>{bill.han_thanh_toan?.split("-").reverse().join("/")}</dd>
          </div>
        </dl>
        {image && (
          <a
            className="button payment-download"
            href={image}
            download={`VietQR-${bill.so_hoa_don}.png`}
          >
            Tải mã QR
          </a>
        )}
        <p className="payment-notice">
          Kiểm tra người nhận, số tiền và giữ đúng nội dung chuyển khoản. Ban
          quản lý sẽ xác nhận sau khi đối soát giao dịch.
        </p>
      </div>
    </section>
  );
}
```

## Thử từ đầu

1. Đăng nhập BQL → **Căn hộ** → tạo căn. Lưu username/mật khẩu trong khung thông tin mới hiện. Có thể đăng nhập tài khoản căn ở trình duyệt khác bằng ô **Số điện thoại hoặc tài khoản căn hộ**.
2. BQL thêm cư dân của căn, chọn quan hệ **Chủ hộ**. Tài khoản cá nhân có thể để trống. Sau khi lưu, tài khoản căn được gắn cho người đó, hệ thống hiện mật khẩu mới. Mật khẩu/phiên vừa cấp lúc tạo căn không còn dùng được.
3. Đăng nhập tài khoản căn bằng mật khẩu mới sẽ vào **Hóa đơn của tôi**. Mở hóa đơn đã gửi, kiểm tra người phụ trách hiện tại, tổng tiền, thông tin ngân hàng và QR.
4. BQL thêm chủ hộ mới nếu cần bàn giao: chọn xác nhận bàn giao trước khi lưu. Tên đăng nhập giữ nguyên theo căn, mật khẩu đổi, phiên của chủ hộ trước hết hiệu lực. Hồ sơ và tài khoản cá nhân của người cũ không bị xóa.
5. Với chủ hộ đã có sẵn, vào **Căn hộ → Xem** để chọn **Chủ hộ nhận tài khoản** và bàn giao. Người được chọn phải đang có quan hệ CHU_HO tại đúng căn. Nếu chỉ quên mật khẩu thì dùng cấp lại, không tạo tài khoản trùng.
6. Ghi nhận chủ hộ phụ trách chuyển đi: tài khoản căn cũ bị khóa. Nếu cần sử dụng khi căn chưa gắn người phụ trách, BQL có thể cấp lại mật khẩu tại Căn hộ; không mở lại bằng mật khẩu đã bàn giao trước đó.

Lưu ý: hóa đơn nháp/hủy vẫn không hiện cho cư dân. Người đang phụ trách tài khoản không tự xác nhận tiền đã vào tài khoản BQL; phần đối soát/ghi nhận thu giữ nguyên.

## Kiểm tra kỹ thuật

Kiểm thử bao gồm tạo tài khoản tự động/không trùng, mật khẩu chỉ trả một lần, gắn chủ hộ và xác nhận bàn giao, phiên cũ hết hiệu lực, giới hạn hóa đơn theo căn, đổi mật khẩu, reset đồng thời, reset đua với chuyển đi và rollback toàn bộ khi ghi audit thất bại. Luồng trình duyệt kiểm tra ở 1440px và 375px, gồm việc khung mật khẩu không bị mất sau khi tải lại dữ liệu danh sách.
